#!/bin/bash
BASE_URL="http://localhost:3002/api/vehicles"

echo "1. POST create vehicle"
curl -s -X POST $BASE_URL -H "Content-Type: application/json" -d '{"ownerId":1,"modelId":1,"plateNumber":"TEST-001","color":"Black","vintage":2024,"mileage":0,"createdBy":"test"}' | jq .

echo "2. GET all vehicles"
curl -s "$BASE_URL" | jq .

echo "3. GET by plate"
curl -s "$BASE_URL/plate/TEST-001" | jq .

echo "4. PUT update"
curl -s -X PUT $BASE_URL/1 -H "Content-Type: application/json" -d '{"mileage":100}' | jq .

echo "5. DELETE vehicle"
curl -s -X DELETE $BASE_URL/1 | jq .
